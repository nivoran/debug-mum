✅ Corrigé Annoté

1. Afficher tous les clients 
SELECT * FROM Client;
- SELECT * : récupère toutes les colonnes
- FROM Client : table utilisée
👉 Requête de base
2. Clients de Antananarivo 
SELECT * FROM Client
WHERE ville = 'Antananarivo';
- WHERE : filtre les lignes
👉 Sélection d’une ville spécifique
3. Trier les produits par prix décroissant 
SELECT * FROM Produit
ORDER BY prix DESC;
- ORDER BY : tri
- DESC : ordre décroissant
4. Produits Informatique 
SELECT P.nom_produit
FROM Produit P
JOIN Categorie C ON P.id_categorie = C.id_categorie
WHERE C.nom_categorie = 'Informatique';
- JOIN : relie les tables
- WHERE : filtre la catégorie
8. Total de chaque commande 
SELECT C.id_commande,
       SUM(P.prix * L.quantite) AS total
FROM Commande C
JOIN Ligne_Commande L ON C.id_commande = L.id_commande
JOIN Produit P ON L.id_produit = P.id_produit
GROUP BY C.id_commande;
- SUM() : addition des montants
- GROUP BY : regroupement par commande
12. Clients sans commande 
SELECT Cl.nom
FROM Client Cl
LEFT JOIN Commande C ON Cl.id_client = C.id_client
WHERE C.id_commande IS NULL;
- LEFT JOIN : garde tous les clients
- IS NULL : aucune commande associée
