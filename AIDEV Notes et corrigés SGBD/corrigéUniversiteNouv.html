✅ Corrigé Annoté

1. Afficher tous les étudiants 
SELECT * FROM Etudiant;
- SELECT * : récupère toutes les colonnes
- FROM Etudiant : table utilisée
👉 Requête de base
2. Étudiants de Antananarivo 
SELECT * FROM Etudiant
WHERE ville = 'Antananarivo';
- WHERE : filtre les lignes
👉 Sélection d’une condition précise
3. Trier les étudiants par nom 
SELECT * FROM Etudiant
ORDER BY nom ASC;
- ORDER BY : tri des résultats
- ASC : ordre alphabétique
4. Cours Informatique 
SELECT C.nom_cours
FROM Cours C
JOIN Formation F ON C.id_formation = F.id_formation
WHERE F.nom_formation = 'Informatique';
- JOIN : relie les tables
- ON : condition de jointure
8. Moyenne par étudiant 
SELECT E.nom, AVG(N.note) AS moyenne
FROM Etudiant E
JOIN Note N ON E.id_etudiant = N.id_etudiant
GROUP BY E.nom;
- AVG() : calcule une moyenne
- GROUP BY : regroupement
12. Étudiants sans note 
SELECT E.nom
FROM Etudiant E
LEFT JOIN Note N ON E.id_etudiant = N.id_etudiant
WHERE N.id_note IS NULL;
- LEFT JOIN : garde tous les étudiants
- IS NULL : absence de correspondance
